\[
\operatorname{Hit@3}=\frac{1}{N}\sum_{j=1}^{N}\mathbf{1}\!\left[\rho_j\leq3\right]
\]

% mrr
\[
\operatorname{MRR@3}=\frac{1}{N}\sum_{j=1}^{N}\begin{cases}\dfrac{1}{\rho_j},&\rho_j\leq3,\\[3pt]0,&\rho_j>3.\end{cases}
\]
